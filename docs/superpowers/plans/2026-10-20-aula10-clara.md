# Aula 10 — CLARA multiagente: plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Montar, publicar e testar no n8n local os dois workflows da Aula 10 (ingestão dos convênios e CLARA multiagente) e produzir a entrega (`workflows/*.json`, `testes.md`, `.zip`).

**Architecture:** Os workflows são descritos em Python (`scripts/wf_*.py`) como dicionários no formato do n8n, com os prompts em `prompts/*.txt`; scripts pequenos, só com a biblioteca padrão, os implantam pela API pública do n8n, enviam os PDFs ao formulário, rodam os cenários da §6.2 pelo webhook do chat e leem o caminho de cada resposta no registro da execução. A apostila é a fonte de verdade de nomes de nó, parâmetros e textos.

**Tech Stack:** n8n 2.42.5 (Docker, `127.0.0.1:5678`), Ollama + `nomic-embed-text`, Groq (`openai/gpt-oss-120b` e `openai/gpt-oss-20b`), Python 3.14 só com stdlib (`urllib`, `json`, `unittest`, `zipfile`), git.

**Spec:** `docs/superpowers/specs/2026-10-20-aula10-clara-design.md`

## Global Constraints

- Nomes de nó exatamente como na apostila e **sem acento nos nós citados como ferramenta**: `Sub-agente INFO`, `Sub-agente CONVENIOS`, `Sub-agente ATENDIMENTO`, `Base de convenios`, `Registrar chamado`.
- Nomes dos workflows: `Aula 10 - Ingestão dos convênios` e `Aula 10 - CLARA multiagente`.
- Memory Key da base vetorial: `convenios`, a mesma nos dois workflows.
- Embeddings: `nomic-embed-text:latest` nos dois workflows; trechos de 1000 com sobreposição de 150; top-k 4.
- Modelos: CLARA `openai/gpt-oss-120b`; sub-agentes `openai/gpt-oss-20b`; temperatura 0.2 em ambos; Max Iterations 4 nos sub-agentes; Memória com janela 10.
- Credenciais no n8n: `Groq` (tipo `groqApi`) e `Ollama` (tipo `ollamaApi`, Base URL `http://host.docker.internal:11434`).
- Data table `chamados` com colunas texto `protocolo`, `nome`, `email`, `descricao`, `registrado_em`.
- Segredos só no `.env` (`N8N_API_KEY`, `GROQ_API_KEY`); nenhum script imprime chave, e nenhum arquivo versionado contém chave.
- Intervalo de 20 s entre mensagens ao chat; em erro do Groq, esperar 60 s e reenviar uma vez.
- Alteração de prompt motivada por cenário com falha **só com aprovação do usuário**.
- O chat público só fica ligado durante os testes; os JSON exportados têm `public` desligado.
- Comandos rodam a partir da raiz do repositório. No Git Bash, `export MSYS_NO_PATHCONV=1` antes de `docker exec` com caminhos absolutos.
- Testes: `python -m unittest discover -s tests -t . -v`.

```python
"""Cliente mínimo da API pública do n8n, com as chaves lidas do .env."""
import json
import os
import urllib.error
import urllib.request
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent
BASE_URL = os.environ.get("N8N_URL", "http://localhost:5678")


class ErroApi(Exception):
    pass


def carregar_env(caminho=RAIZ / ".env"):
    valores = {}
    for linha in Path(caminho).read_text(encoding="utf-8").splitlines():
        linha = linha.strip()
        if not linha or linha.startswith("#") or "=" not in linha:
            continue
        chave, valor = linha.split("=", 1)
        valores[chave.strip()] = valor.strip().strip('"').strip("'")
    return valores


def api(metodo, caminho, corpo=None):
    env = carregar_env()
    dados = None if corpo is None else json.dumps(corpo).encode("utf-8")
    requisicao = urllib.request.Request(
        f"{BASE_URL}/api/v1{caminho}",
        data=dados,
        method=metodo,
        headers={
            "X-N8N-API-KEY": env["N8N_API_KEY"],
            "Content-Type": "application/json",
            "Accept": "application/json",
        },
    )
    try:
        with urllib.request.urlopen(requisicao, timeout=60) as resposta:
            texto = resposta.read().decode("utf-8")
    except urllib.error.HTTPError as erro:
        detalhe = erro.read().decode("utf-8", "replace")[:500]
        raise ErroApi(f"{metodo} {caminho} -> {erro.code}: {detalhe}") from None
    return json.loads(texto) if texto else None
```

```python
"""Cria no n8n, se ainda não existirem, as credenciais Groq e Ollama e a Data table chamados."""
from scripts.n8n_api import ErroApi, api, carregar_env

OLLAMA_URL = "http://host.docker.internal:11434"
COLUNAS = ("protocolo", "nome", "email", "descricao", "registrado_em")


def _buscar_credencial(nome, tipo):
    for credencial in api("GET", "/credentials?limit=250")["data"]:
        if credencial["name"] == nome and credencial["type"] == tipo:
            return credencial["id"]
    return None


def _buscar_tabela(nome):
    for tabela in api("GET", "/data-tables?limit=100")["data"]:
        if tabela["name"] == nome:
            return tabela["id"]
    return None


def obter_ids():
    ids = {
        "groq": _buscar_credencial("Groq", "groqApi"),
        "ollama": _buscar_credencial("Ollama", "ollamaApi"),
        "chamados": _buscar_tabela("chamados"),
    }
    faltando = [chave for chave, valor in ids.items() if not valor]
    if faltando:
        raise ErroApi(f"Recursos ausentes no n8n: {faltando}. Rode python -m scripts.configurar_n8n")
    return ids


def _garantir_credencial(nome, tipo, dados):
    existente = _buscar_credencial(nome, tipo)
    if existente:
        return existente, False
    return api("POST", "/credentials", {"name": nome, "type": tipo, "data": dados})["id"], True


def _garantir_tabela():
    existente = _buscar_tabela("chamados")
    if existente:
        return existente, False
    corpo = {"name": "chamados", "columns": [{"name": c, "type": "string"} for c in COLUNAS]}
    return api("POST", "/data-tables", corpo)["id"], True


def main():
    env = carregar_env()
    recursos = {
        "Groq": _garantir_credencial("Groq", "groqApi", {"apiKey": env["GROQ_API_KEY"]}),
        "Ollama": _garantir_credencial("Ollama", "ollamaApi", {"baseUrl": OLLAMA_URL}),
        "chamados": _garantir_tabela(),
    }
    for nome, (ident, criado) in recursos.items():
        print(f"{nome}: {ident} ({'criado' if criado else 'já existia'})")
    for nome in ("Groq", "Ollama"):
        try:
            resultado = api("POST", f"/credentials/{recursos[nome][0]}/test")
            print(f"teste {nome}: {resultado}")
        except ErroApi as erro:
            print(f"teste {nome}: FALHOU {erro}")


if __name__ == "__main__":
    main()
```

```python
"""Peças comuns para descrever workflows do n8n como dicionários."""
import re
import uuid


def no(nome, tipo, versao, posicao, parametros, credenciais=None, webhook_id=None):
    definicao = {
        "parameters": parametros,
        "id": str(uuid.uuid5(uuid.NAMESPACE_URL, f"aula10/{tipo}/{nome}")),
        "name": nome,
        "type": tipo,
        "typeVersion": versao,
        "position": list(posicao),
    }
    if credenciais:
        definicao["credentials"] = credenciais
    if webhook_id:
        definicao["webhookId"] = webhook_id
    return definicao


def ligar(conexoes, origem, destino, tipo="main"):
    ramos = conexoes.setdefault(origem, {}).setdefault(tipo, [[]])
    ramos[0].append({"node": destino, "type": tipo, "index": 0})


def nome_ferramenta(nome_no):
    # Regra do n8n (§3.2): tudo que não é letra sem acento, algarismo, hífen ou sublinhado vira _.
    return re.sub(r"[^A-Za-z0-9_-]", "_", nome_no)


def cred_groq(ids):
    return {"groqApi": {"id": ids["groq"], "name": "Groq"}}


def cred_ollama(ids):
    return {"ollamaApi": {"id": ids["ollama"], "name": "Ollama"}}
```

```python
"""Leitura das execuções de um workflow pela API pública (equivale ao painel Logs)."""
import time

from scripts.n8n_api import api

TERMINADAS = {"success", "error", "crashed", "canceled"}


def ultimo_id(workflow_id):
    dados = api("GET", f"/executions?workflowId={workflow_id}&limit=1")["data"]
    return int(dados[0]["id"]) if dados else 0


def esperar_execucao(workflow_id, depois_de, limite_s=120):
    fim = time.time() + limite_s
    while time.time() < fim:
        dados = api("GET", f"/executions?workflowId={workflow_id}&limit=1")["data"]
        if dados and int(dados[0]["id"]) > depois_de and dados[0]["status"] in TERMINADAS:
            return api("GET", f"/executions/{dados[0]['id']}?includeData=true")
        time.sleep(2)
    raise TimeoutError(f"nenhuma execução nova de {workflow_id} terminou em {limite_s} s")


def nos_executados(execucao):
    return list(execucao["data"]["resultData"]["runData"].keys())


def itens_saida(execucao, nome_no):
    total = 0
    for rodada in execucao["data"]["resultData"]["runData"].get(nome_no, []):
        saidas = (rodada.get("data") or {}).get("main") or []
        if saidas and saidas[0]:
            total += len(saidas[0])
    return total
```

```python
"""Cria ou atualiza um workflow da aula no n8n e o publica.

    for wf in api("GET", "/workflows?limit=250")["data"]:

    else:
        wf = api("POST", "/workflows", definicao)

    else:
        sys.exit(__doc__)
    print(f"{definicao['name']}: {implantar(definicao)} publicado")


if __name__ == "__main__":
    main(sys.argv[1:])
```

```python
"""Envia os PDFs de convenios/ ao formulário publicado e confere a execução (seção 4.7)."""
import re
import urllib.request
import uuid

from scripts import wf_ingestao
from scripts.execucoes import esperar_execucao, itens_saida, ultimo_id
from scripts.implantar import buscar_workflow
from scripts.n8n_api import BASE_URL, RAIZ

URL_FORM = f"{BASE_URL}/form/base-convenios"


def nome_campo_arquivo(html):
    for tag in re.findall(r"<input\b[^>]*>", html):
        if re.search(r'type="file"', tag):
            return re.search(r'name="([^"]+)"', tag).group(1)
    raise ValueError("o formulário não tem campo de arquivo")


def montar_multipart(campo, arquivos, limite):
    partes = []
    for nome, conteudo in arquivos:
        cabecalho = (
            f'--{limite}\r\nContent-Disposition: form-data; name="{campo}"; filename="{nome}"\r\n'
            "Content-Type: application/pdf\r\n\r\n"
        )
        partes.append(cabecalho.encode("utf-8") + conteudo + b"\r\n")
    return b"".join(partes) + f"--{limite}--\r\n".encode("utf-8")


def main():
    workflow_id = buscar_workflow(wf_ingestao.NOME)["id"]
    antes = ultimo_id(workflow_id)
    with urllib.request.urlopen(URL_FORM, timeout=30) as resposta:
        campo = nome_campo_arquivo(resposta.read().decode("utf-8"))
    pdfs = sorted((RAIZ / "convenios").glob("*.pdf"))
    limite = uuid.uuid4().hex
    corpo = montar_multipart(campo, [(p.name, p.read_bytes()) for p in pdfs], limite)
    requisicao = urllib.request.Request(URL_FORM, data=corpo, method="POST",
                                        headers={"Content-Type": f"multipart/form-data; boundary={limite}"})
    with urllib.request.urlopen(requisicao, timeout=120) as resposta:
        print(f"POST {URL_FORM} -> {resposta.status}")
    execucao = esperar_execucao(workflow_id, antes)
    arquivos = itens_saida(execucao, "Um item por arquivo")
    trechos = itens_saida(execucao, "Inserir na base")
    print(f"execução {execucao['id']}: {execucao['status']}")
    print(f"{arquivos} arquivo(s) indexado(s) em {trechos} trecho(s).")
    if execucao["status"] != "success" or (arquivos, trechos) != (len(pdfs), len(pdfs)):
        raise SystemExit("ingestão não conferiu: esperado um trecho por PDF")


if __name__ == "__main__":
    main()
```

```python
"""Envia mensagens ao chat público da CLARA (webhook do Chat Trigger)."""
import json
import urllib.error
import urllib.request

from scripts import wf_clara
from scripts.n8n_api import BASE_URL

URL_CHAT = f"{BASE_URL}/webhook/{wf_clara.WEBHOOK_CHAT}/chat"


class ErroChat(Exception):
    pass


def enviar(session_id, mensagem):
    corpo = json.dumps({"action": "sendMessage", "sessionId": session_id, "chatInput": mensagem}).encode("utf-8")
    requisicao = urllib.request.Request(URL_CHAT, data=corpo, method="POST",
                                        headers={"Content-Type": "application/json"})
    try:
        with urllib.request.urlopen(requisicao, timeout=180) as resposta:
            texto = resposta.read().decode("utf-8")
    except urllib.error.HTTPError as erro:
        raise ErroChat(f"{erro.code}: {erro.read().decode('utf-8', 'replace')[:300]}") from None
    try:
        return json.loads(texto)["output"]
    except (ValueError, KeyError):
        return texto
```

```python
def celula(texto):
    return str(texto).replace("|", "\\|").replace("\r", "").replace("\n", "<br>")


def gerar(resultados, ajustes=()):
    linhas = []
    for r in resultados:
        sub = ", ".join(s.replace("Sub-agente ", "") for s in r["acionados"]) or "nenhum"
        ok = "sim" if r["ok"] else "não — " + "; ".join(r["motivos"])
        linhas.append(f"| {r['id']} | {celula(r['mensagem'])} | {celula(r['resposta'])} | {sub} | "
                      f"{celula(r['esperado'])} | {celula(ok)} |")
    corpo = CABECALHO + "\n".join(linhas) + "\n" + AJUSTES
    if ajustes:
        corpo += "\n".join(f"| {a['cenario']} | {celula(a['anterior'])} | {celula(a['nova'])} | "
                           f"{celula(a['resultado'])} |" for a in ajustes) + "\n"
    else:
        corpo += "| — | Nenhum ajuste necessário | — | — |\n"
    return corpo
```

```python
"""Desliga o chat público, exporta os dois workflows para workflows/ e gera o .zip de entrega."""
import json
import re
import zipfile

from scripts import wf_clara, wf_ingestao
from scripts.configurar_n8n import obter_ids
from scripts.implantar import buscar_workflow, implantar
from scripts.n8n_api import RAIZ, api

CAMPOS = ("name", "nodes", "connections", "settings", "pinData", "meta")
ARQUIVOS = {wf_ingestao.NOME: "ingestao-convenios.json", wf_clara.NOME: "clara-multiagente.json"}
ZIP = RAIZ / "entrega-aula10.zip"


def normalizar(wf):
    return {campo: (wf.get(campo) or {}) if campo in ("pinData", "meta") else wf.get(campo) for campo in CAMPOS}


def sem_segredos(wf):
    texto = json.dumps(wf)
    return not re.search(r'"(apiKey|password|accessToken|token)"\s*:|gsk_[A-Za-z0-9]', texto)


def main():
    implantar(wf_clara.construir(obter_ids(), chat_publico=False))
    for nome, arquivo in ARQUIVOS.items():
        wf = normalizar(api("GET", f"/workflows/{buscar_workflow(nome)['id']}"))
        if not sem_segredos(wf):
            raise SystemExit(f"{nome}: possível segredo no JSON; exportação interrompida")
        destino = RAIZ / "workflows" / arquivo
        destino.write_text(json.dumps(wf, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
        print(f"{nome} -> {destino.relative_to(RAIZ)}")
    with zipfile.ZipFile(ZIP, "w", zipfile.ZIP_DEFLATED) as pacote:
        for arquivo in ARQUIVOS.values():
            pacote.write(RAIZ / "workflows" / arquivo, f"workflows/{arquivo}")
        pacote.write(RAIZ / "testes.md", "testes.md")
    print(f"pacote: {ZIP.name}")


if __name__ == "__main__":
    main()
```
